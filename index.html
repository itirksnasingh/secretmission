R PROGRAMMING / EXPERIMENT 08

Classification using the Iris dataset

Reference material from your uploaded R Programming Practical Code Book.

8.1

Prepare Iris data and create train/test split

Uses the built-in Iris dataset and creates a reproducible 70/30 train/test split.

data(iris)
iris_data <- iris
iris_data$Species <- factor(iris_data$Species)

set.seed(123)
train_index <- sample(
    seq_len(nrow(iris_data)),
    size = floor(0.70 * nrow(iris_data))
)

train_data <- iris_data[train_index, ]
test_data <- iris_data[-train_index, ]

str(iris_data)
summary(iris_data)
8.2

Logistic regression: setosa vs non-setosa

Fits a binary logistic regression model for setosa versus non-setosa.

train_log <- train_data
test_log <- test_data

train_log$IsSetosa <- ifelse(train_log$Species == "setosa", 1, 0)
test_log$IsSetosa <- ifelse(test_log$Species == "setosa", 1, 0)

log_model <- glm(
    IsSetosa ~ Sepal.Length + Sepal.Width +
        Petal.Length + Petal.Width,
    data = train_log,
    family = binomial
)

log_prob <- predict(log_model, newdata = test_log, type = "response")
log_pred <- ifelse(log_prob >= 0.5, 1, 0)

table(Actual = test_log$IsSetosa, Predicted = log_pred)
mean(log_pred == test_log$IsSetosa)
8.3

Decision tree classification

Trains a decision tree to predict all three Iris species; requires rpart and rpart.plot.

# Install once if needed:
# install.packages(c("rpart", "rpart.plot"))
library(rpart)
library(rpart.plot)

tree_model <- rpart(
    Species ~ Sepal.Length + Sepal.Width +
        Petal.Length + Petal.Width,
    data = train_data,
    method = "class"
)

rpart.plot(tree_model)

tree_pred <- predict(tree_model, newdata = test_data, type = "class")
tree_cm <- table(Actual = test_data$Species, Predicted = tree_pred)
print(tree_cm)
tree_accuracy <- mean(tree_pred == test_data$Species)
print(tree_accuracy)
8.4

kNN classification with normalized predictors

Runs k-nearest neighbours with normalized predictors and k = 5.

# install.packages("class") # usually included with R
library(class)

features <- c("Sepal.Length", "Sepal.Width",
              "Petal.Length", "Petal.Width")

train_x <- train_data[, features]
test_x <- test_data[, features]

# Normalize using training-set mean and standard deviation.
train_means <- sapply(train_x, mean)
train_sds <- sapply(train_x, sd)
train_sds[train_sds == 0] <- 1

train_scaled <- scale(train_x, center = train_means, scale = train_sds)
test_scaled <- scale(test_x, center = train_means, scale = train_sds)

knn_pred <- knn(
    train = train_scaled,
    test = test_scaled,
    cl = train_data$Species,
    k = 5
)

knn_cm <- table(Actual = test_data$Species, Predicted = knn_pred)
print(knn_cm)
knn_accuracy <- mean(knn_pred == test_data$Species)
print(knn_accuracy)
8.5

Compare model accuracy

Compares displayed accuracy values; logistic regression is binary while tree and kNN classify all three species, as noted in the source.

print(paste("Decision Tree Accuracy:", round(tree_accuracy, 3)))
print(paste("kNN Accuracy:", round(knn_accuracy, 3)))
print(paste("Logistic Regression Accuracy:",
            round(mean(log_pred == test_log$IsSetosa), 3)))

9.4

Run K-Means for k = 3, 4, and 5
